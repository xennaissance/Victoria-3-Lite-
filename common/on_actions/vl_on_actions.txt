# Victoria 3 Lite on_actions.
# Hooks into on_game_started (fires once when a new game is set up, no root scope)
# through its on_actions list, so vanilla's own effect block is left untouched.

on_game_started = {
    on_actions = {
        vl_set_name_anchor_on_action
        vl_unify_culture_on_action
        vl_unify_religion_on_action
        vl_consolidate_pops_on_action
        vl_track_insurgents_on_action
        vl_state_cache_on_action
    }
}

# Insurgent-country list (see below) is rebuilt every month and whenever a revolution starts.
# The state cache is built once a year; the monthly hook only builds it if it is missing
# (a save from before this version, or one where it was never built).
on_monthly_pulse = {
    on_actions = {
        vl_track_insurgents_on_action
        vl_state_cache_if_missing_on_action
    }
}

on_yearly_pulse = {
    on_actions = {
        vl_state_cache_on_action
    }
}

on_revolution_start = {
    on_actions = {
        vl_track_insurgents_on_action
    }
}

# Culture and religion names that follow the game rules are customizable localization, which must be
# called on a valid object. A global variable holding the Human culture (which always exists) is that object.
vl_set_name_anchor_on_action = {
    effect = {
        set_global_variable = {
            name = vl_name_anchor
            value = cu:vl_human
        }
    }
}

# Pop Cultures rule.
# One culture: every country gets Human as its only primary culture, and every pop becomes Human.
# Heritage groups / Heritages: cultures merge into one existing culture per group or heritage, which
# also inherits the merged cultures' homelands (see scripted_effects/vl_culture_effects.txt).
vl_unify_culture_on_action = {
    effect = {
        if = {
            limit = { has_game_rule = vl_one_culture }
            every_country = {
                save_scope_as = vl_country
                add_primary_culture = cu:vl_human
                every_primary_culture = {
                    limit = { NOT = { this = cu:vl_human } }
                    save_scope_as = vl_old_culture
                    scope:vl_country = {
                        remove_primary_culture = scope:vl_old_culture
                    }
                }
            }
            every_state = {
                every_scope_pop = {
                    change_pop_culture = {
                        target = cu:vl_human
                        value = 1
                    }
                }
            }
        }
        else_if = {
            limit = { has_game_rule = vl_culture_groups }
            vl_homelands_to_group_effect = yes
            every_country = { vl_country_cultures_to_group_effect = yes }
            every_state = {
                every_scope_pop = { vl_pop_culture_to_group_effect = yes }
            }
        }
        else_if = {
            limit = { has_game_rule = vl_culture_heritages }
            vl_homelands_to_heritage_effect = yes
            every_country = { vl_country_cultures_to_heritage_effect = yes }
            every_state = {
                every_scope_pop = { vl_pop_culture_to_heritage_effect = yes }
            }
        }
    }
}

# Pop Religions rule. One faith: everyone follows the One Faith. Heritage groups and Heritages:
# religions merge into one existing religion per group or heritage (see scripted_effects/vl_religion_effects.txt).
vl_unify_religion_on_action = {
    effect = {
        if = {
            limit = { has_game_rule = vl_religion_one_faith }
            every_country = {
                set_state_religion = rel:vl_faith
            }
            every_state = {
                every_scope_pop = {
                    change_pop_religion = {
                        target = rel:vl_faith
                        value = 1
                    }
                }
            }
        }
        else_if = {
            limit = { has_game_rule = vl_religion_groups }
            every_country = { vl_country_religion_to_group_effect = yes }
            every_state = {
                every_scope_pop = { vl_pop_religion_to_group_effect = yes }
            }
        }
        else_if = {
            limit = { has_game_rule = vl_religion_heritages }
            every_country = { vl_country_religion_to_heritage_effect = yes }
            every_state = {
                every_scope_pop = { vl_pop_religion_to_heritage_effect = yes }
            }
        }
    }
}

# Pop Consolidation rule. Runs after the culture and religion rules: in every state, all pops
# take the state's largest culture and largest religion (by population), so pops merge.
# Countries' primary cultures and state religions are left as they are.
vl_consolidate_pops_on_action = {
    effect = {
        if = {
            limit = { has_game_rule = vl_consolidation_on }
            every_state = {
                ordered_scope_pop = {
                    order_by = vl_state_size_of_pop_culture
                    culture = { save_scope_as = vl_top_culture }
                }
                ordered_scope_pop = {
                    order_by = vl_state_size_of_pop_religion
                    religion = { save_scope_as = vl_top_religion }
                }
                every_scope_pop = {
                    limit = { NOT = { culture = scope:vl_top_culture } }
                    change_pop_culture = {
                        target = scope:vl_top_culture
                        value = 1
                    }
                }
                every_scope_pop = {
                    limit = { NOT = { religion = scope:vl_top_religion } }
                    change_pop_religion = {
                        target = scope:vl_top_religion
                        value = 1
                    }
                }
            }
        }
    }
}

# Countries with at least one insurrectionary movement. Vanilla's cultural and religious movements
# scanned every neighbouring state of their country for an insurrection, for every movement;
# Lite's patched movements (common/political_movements/vl_movement_overrides.txt) check this short list instead.
vl_track_insurgents_on_action = {
    effect = {
        clear_global_variable_list = vl_insurgent_countries
        every_country = {
            limit = {
                any_political_movement = { is_insurrectionary = yes }
            }
            add_to_global_variable_list = {
                name = vl_insurgent_countries
                target = this
            }
        }
    }
}

# State cache (memoization). The AI's state_value (ai_strategies/vl_ai_strategy_inject.txt) asked every
# country, for every state, whether the state has a potential strait province and rubber, oil or gold
# potential. Those depend only on the state, so they are stored here once a year as state variables
# (used as flags). vl_state_cache_ready tells state_value the cache exists; before it does (an older save,
# until its first monthly pulse) vanilla's own checks run.
vl_state_cache_on_action = {
    effect = {
        every_state = {
            if = {
                limit = { has_any_potential_strait_province = yes }
                set_variable = vl_strait_any
            }
            else_if = {
                limit = { has_variable = vl_strait_any }
                remove_variable = vl_strait_any
            }
            if = {
                limit = {
                    OR = {
                        "num_potential_resources(bt:building_rubber_plantation)" > 0
                        "num_potential_resources(bt:building_oil_rig)" > 0
                        "num_potential_resources(bt:building_gold_field)" > 0
                        "num_potential_resources(bt:building_gold_mine)" > 0
                    }
                }
                set_variable = vl_valuable_resources
            }
            else_if = {
                limit = { has_variable = vl_valuable_resources }
                remove_variable = vl_valuable_resources
            }
        }
        set_global_variable = vl_state_cache_ready
    }
}

vl_state_cache_if_missing_on_action = {
    trigger = { NOT = { has_global_variable = vl_state_cache_ready } }
    on_actions = { vl_state_cache_on_action }
}
