# Victoria 3 Lite: cheaper infamy of a whole target country (used by treaty-article demands and several war goals,
# where it is multiplied by a small severity and clamped to a minimum and maximum).
# Vanilla runs the full per-state infamy calculation for every state of the target. That stays for major and
# great powers (including unrecognized major powers), whose numbers players see. When a smaller country is the
# aggressor, a country-wide estimate from its share of world population and GDP is used instead.
# root = the aggressor country, this = the target country (as in vanilla).

REPLACE:country_infamy_value = {
	value = 0

	if = {
		limit = {
			root = { country_rank >= rank_value:unrecognized_major_power }
		}
		add = {
			desc = "INFAMY_COUNTRY_TOTAL"
			every_scope_state = {
				add = state_infamy_value
			}
		}
	}
	else = {
		add = {
			desc = "INFAMY_COUNTRY_TOTAL"
			add = {
				value = total_population
				divide = {
					value = global_population
					min = 1
				}
				multiply = define:NDiplomacy|WAR_GOAL_INFAMY_POPULATION_WEIGHT
			}
			add = {
				value = gdp
				divide = {
					value = global_gdp
					min = 1
				}
				multiply = define:NDiplomacy|WAR_GOAL_INFAMY_GDP_WEIGHT
			}
			multiply = define:NDiplomacy|WAR_GOAL_INFAMY_WORLD_SHARE_SCALE

			# With One culture every state is a homeland of the aggressor, which vanilla discounts
			if = {
				limit = { has_game_rule = vl_one_culture }
				multiply = {
					value = 1
					add = define:NDiplomacy|WAR_GOAL_INFAMY_HOMELAND_FACTOR
				}
			}

			# Vanilla counts every state at least WAR_GOAL_INFAMY_STATE_BASELINE (1) for population and for GDP
			min = 2
		}
	}
}
