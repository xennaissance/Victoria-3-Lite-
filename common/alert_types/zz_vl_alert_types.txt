# Victoria 3 Lite: cheaper alerts.
# Alerts are re-checked constantly for the player. The "low expected standard of living in state"
# check costs about 0.7 s of worst-thread time per real minute in a late game (profiled 13 Nov 2026).
# It is skipped unless the "Lite: Costly Alerts" game rule is set to keep it.
#
# This file relies on implicit replace (a later file redefining the same key). If the alert still
# shows with the rule at its default, or error.log reports a duplicate key, fall back to overwriting
# the whole vanilla 00_alert_types.txt.

low_expected_sol_in_state = {
    texture = "gfx/interface/icons/alert_icons/starving.dds"
    script_context = player_state

    valid = {
        has_game_rule = vl_costly_alerts_on
        population_below_expected_sol >= 0.25
    }

    open_panel = states_panel
    type = important_action
    color = yellow
    alert_group = low_sol_in_state
}
