# Victoria 3 Lite: religion merging used by the "Pop Religions" game rule.
# Heritage groups: Abrahamic -> Catholic, Eastern -> Confucian, Naturalistic -> Animist, Non-spiritual -> Atheist.
# Heritages: Christian -> Catholic, Islamic -> Sunni, Dharmic -> Hindu, Taoic -> Confucian,
#            Jewish, Indigenous (Animist) and Materialist (Atheist) stay as they are.
# The surviving religions are renamed on screen to match (see customizable_localization/vl_religion_names.txt).

vl_pop_religion_to_group_effect = {
    if = {
        limit = { religion = { has_discrimination_trait_group = heritage_group_abrahamic } }
        change_pop_religion = { target = rel:catholic value = 1 }
    }
    else_if = {
        limit = { religion = { has_discrimination_trait_group = heritage_group_eastern } }
        change_pop_religion = { target = rel:confucian value = 1 }
    }
    else_if = {
        limit = { religion = { has_discrimination_trait_group = heritage_group_naturalistic } }
        change_pop_religion = { target = rel:animist value = 1 }
    }
    else_if = {
        limit = { religion = { has_discrimination_trait_group = heritage_group_non_spiritual } }
        change_pop_religion = { target = rel:atheist value = 1 }
    }
}
